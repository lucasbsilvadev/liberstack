# snapshot.py
from pathlib import Path

IGNORE_DIRS = {
    ".venv", "venv", ".env", "env",
    "__pycache__", ".pytest_cache", ".mypy_cache", ".ruff_cache",
    ".git", ".idea", ".vscode",
    "node_modules", "htmlcov", "package-lock.json"
}
IGNORE_FILES = {
    ".coverage", "coverage.xml", "snapshot.txt", "pytest-output.txt",
}
IGNORE_EXTS = {".pyc", ".pyo", ".pyd", ".log"}

def ignored(p: Path) -> bool:
    if p.name in IGNORE_DIRS or p.name in IGNORE_FILES:
        return True
    if p.suffix in IGNORE_EXTS:
        return True
    return False

def tree(root: Path, prefix: str = "") -> list[str]:
    lines = []
    items = sorted(root.iterdir(), key=lambda p: (not p.is_dir(), p.name.lower()))
    for item in items:
        if ignored(item):
            continue
        if item.is_dir():
            lines.append(f"{prefix}+---{item.name}")
            lines.extend(tree(item, prefix + "|   "))
        else:
            lines.append(f"{prefix}|   {item.name}")
    return lines

def dump(root: Path) -> str:
    parts = ["=== ESTRUTURA ===", "Listagem de caminhos de pasta", ""]
    parts.extend(tree(root))
    parts.append("")
    parts.append("")
    parts.append("=== CONTEUDO DOS ARQUIVOS ===")
    parts.append("")

    for path in sorted(root.rglob("*")):
        if path.is_dir() or ignored(path):
            continue
        if any(part in IGNORE_DIRS for part in path.parts):
            continue
        rel = path.relative_to(root)
        parts.append("=" * 60)
        parts.append(f"FILE: {rel}")
        parts.append("=" * 60)
        try:
            parts.append(path.read_text(encoding="utf-8"))
        except UnicodeDecodeError:
            parts.append("[binário — ignorado]")
        parts.append("")

    return "\n".join(parts)

if __name__ == "__main__":
    out = dump(Path("."))
    Path("snapshot.txt").write_text(out, encoding="utf-8")
    print(f"snapshot.txt gerado ({out.count(chr(10))} linhas)")

import React, { useState, useEffect } from 'react';
import { Box, Typography } from '@mui/material';
import { motion, AnimatePresence } from 'framer-motion';
import '../../style/preloader.css';

const Preloader = () => {
  const [progress, setProgress] = useState(0);
  const [isVisible, setIsVisible] = useState(true);

  useEffect(() => {
    // Simula o carregamento
    const timer = setInterval(() => {
      setProgress((oldProgress) => {
        if (oldProgress === 100) {
          clearInterval(timer);
          setTimeout(() => setIsVisible(false), 500);
          return 100;
        }
        return Math.min(oldProgress + Math.random() * 10, 100);
      });
    }, 200);

    return () => {
      clearInterval(timer);
    };
  }, []);

  return (
    <AnimatePresence>
      {isVisible && (
        <motion.div
          initial={{ opacity: 1 }}
          exit={{ 
            opacity: 0,
            transition: { duration: 0.5, ease: "easeOut" }
          }}
        >
          <Box className="preloader-container">
            
            {/* Container da Logo e Barra de Progresso */}
            <Box className="preloader-content">
              
              {/* Logo - Sua imagem com rotação */}
              <motion.div
                initial={{ rotate: 0 }}
                animate={{ rotate: 360 }}
                transition={{ 
                  duration: 2, 
                  repeat: Infinity, 
                  ease: "linear" 
                }}
              >
                <img 
                  src="/assets/libericon-nobg.png" 
                  alt="liberstack" 
                  className="preloader-logo"
                  onError={(e) => {
                    // Fallback caso a imagem não carregue
                    e.target.style.display = 'none';
                    console.log('Imagem não encontrada em /assets/libericon.png');
                  }}
                />
              </motion.div>

              {/* Barra de Progresso Retangular */}
              <Box className="preloader-progress-container">
                
                {/* Barra de progresso */}
                <Box className="preloader-progress-bar">

                >
                  <Typography className="preloader-percentage">
                    {Math.round(progress)}%
                  </Typography>
                </motion.div>
              </Box>

              {/* Texto "liberstack" abaixo da barra */}

// src/components/ui/BinaryRain.js
import React, { useMemo } from 'react';
import { Box } from '@mui/material';
import { motion } from 'framer-motion';

/**
 * Chuva de "0" e "1" para usar como fundo decorativo.
 * Props:
 *  - columns: número de colunas verticais (default 20)
 *  - charsPerColumn: altura de cada coluna em caracteres (default 24)
 *  - opacity: opacidade geral (default 0.18)
 *  - color: cor dos caracteres (default '#7C3AED')
 *  - speed: multiplicador de velocidade (default 1). Menor = mais rápido.
 *  - density: proporção da altura que a coluna preenche (0-1, default 0.7)
 */
const BinaryRain = ({
  columns = 10,
  charsPerColumn = 24,
  opacity = 0.18,
  color = '#7C3AED',
  speed = 1,
  density = 0.7,
}) => {
  // useMemo garante que as colunas só são geradas uma vez por montagem,
  // evitando recriação a cada re-render (performance).
  const cols = useMemo(() => {
    return Array.from({ length: columns }).map((_, i) => {
      const chars = Array.from({ length: charsPerColumn })
        .map(() => (Math.random() > 0.5 ? '1' : '0'))
        .join('\n');
      return {
        id: i,
        chars,
        delay: Math.random() * 5,
        duration: (8 + Math.random() * 8) * speed,
        left: `${(i / columns) * 100}%`,
        charsPerColumn,
      };
    });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [columns, charsPerColumn]);

  return (
    <Box
      aria-hidden="true"
      sx={{
        position: 'absolute',
        inset: 0,
        overflow: 'hidden',
        pointerEvents: 'none',
        opacity,
        color,
        fontFamily: 'monospace',
        fontSize: { xs: '0.55rem', sm: '0.65rem', md: '0.75rem' },
        lineHeight: 1.1,
        userSelect: 'none',
        whiteSpace: 'pre',
      }}
    >
      {cols.map((col) => (
        <motion.div
          key={col.id}
          initial={{ y: '-100%' }}
          animate={{ y: '100%' }}
          transition={{
            duration: col.duration,
            delay: col.delay,
            repeat: Infinity,
            ease: 'linear',
          }}
          style={{
            position: 'absolute',
            top: 0,
            left: col.left,
            height: `${density * 100}%`,
            display: 'flex',
            flexDirection: 'column',
            justifyContent: 'flex-start',
          }}
        >
          {col.chars}
        </motion.div>
      ))}
    </Box>
  );
};

export default BinaryRain;
